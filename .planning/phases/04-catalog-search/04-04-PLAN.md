**src/routes/search.ts — Search page router:**

```typescript
import express from 'express';
import { SearchService } from '../services/search';
import { CatalogService } from '../services/catalog';
const router = express.Router();
const searchService = new SearchService();
const catalogService = new CatalogService();

// Helper: normalize multi-value query params
function toArray(v: any): string[] {
  if (!v) return [];
  return Array.isArray(v) ? v.map(String) : [String(v)];
}

// GET /search
router.get('/', async (req, res, next) => {
  try {
    const q = typeof req.query.q === 'string' ? req.query.q.trim() : '';
    const scope = (req.session?.user?.role === 'CURATOR' || req.session?.user?.role === 'ADMIN')
      ? 'CURATOR' : 'PUBLIC';

    // Parse filter params (same as catalog route)
    const filters = {
      maturity_level: toArray(req.query['maturity_level[]'] || req.query.maturity_level),
      review_status: toArray(req.query['review_status[]'] || req.query.review_status),
      contributing_office: toArray(req.query['contributing_office[]'] || req.query.contributing_office),
      mission_area: toArray(req.query['mission_area[]'] || req.query.mission_area),
      technology_area: toArray(req.query['technology_area[]'] || req.query.technology_area),
      reuse_potential: typeof req.query.reuse_potential === 'string' ? req.query.reuse_potential : undefined,
    };

    const page = Math.max(parseInt(String(req.query.page || '1'), 10) || 1, 1);
    const pageSize = Math.min(Math.max(parseInt(String(req.query.page_size || '12'), 10) || 12, 1), 50);

    // Validate q length BEFORE calling service
    if (q.length > 500) {
      return res.status(400).render('search/index', {
        query: q,
        results: null,
        filters: await catalogService.getFilters(scope),
        userRole: req.session?.user?.role || 'PUBLIC',
        error: 'Your search query is too long. Please shorten it to 500 characters or fewer.',
        activeFilters: filters,
      });
    }

    const [searchResults, filterFacets] = await Promise.all([
      searchService.search(q, filters, page, pageSize, scope),
      catalogService.getFilters(scope),
    ]);

    res.render('search/index', {
      query: q,
      results: searchResults,
      filters: filterFacets,
      userRole: req.session?.user?.role || 'PUBLIC',
      error: null,
      activeFilters: filters,
    });
  } catch (err: any) {
    if (err.code === 'QUERY_TOO_LONG') {
      return res.status(400).render('search/index', {
        query: req.query.q || '',
        results: null,
        filters: {},
        userRole: req.session?.user?.role || 'PUBLIC',
        error: 'Your search query is too long. Please shorten it to 500 characters or fewer.',
        activeFilters: {},
      });
    }
    next(err);
  }
});

export { router as searchPageRouter };
```

```html
          <% if (record.mission_area_tags.length > 3) { %>

          <% } %>

</article>
```

SECURITY NOTE on `<%- record.highlight %>`: ts_headline is generated by PostgreSQL and contains only `<b>` or `<em>` HTML tags wrapping matched lexemes — it is NOT user-entered HTML and cannot contain script injection. The highlight field never passes through user input; it is computed by PostgreSQL from the record's own stored text. Using `<%-` (unescaped) for this field is intentional and safe. However, the record.highlight value is validated to contain only safe markup: SearchService must sanitize the ts_headline output to allow only `<b>`, `<em>`, `<strong>` tags (strip any unexpected HTML). Add this to SearchService:
```typescript
private sanitizeHighlight(html: string): string {
  // Allow only <b>, <em>, <strong> from ts_headline; strip everything else
  return html.replace(/<(?!\/?(?:b|em|strong)\b)[^>]*>/gi, '');
}
```
Apply sanitizeHighlight to each record.highlight before returning from search().
