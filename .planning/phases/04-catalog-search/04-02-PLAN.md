**src/routes/catalog.ts — Public catalog page router:**

```typescript
import express from 'express';
import { CatalogService } from '../services/catalog';
const router = express.Router();
const catalogService = new CatalogService();

// Catalog IS the root — both / and /catalog
router.get(['/', '/catalog'], async (req, res, next) => {
  try {
    // Initial server-side render: fetch first page of unfiltered published records
    const scope = (req.session?.user?.role === 'CURATOR' || req.session?.user?.role === 'ADMIN')
      ? 'CURATOR' : 'PUBLIC';
    const [catalogPage, filters] = await Promise.all([
      catalogService.getCatalog({}, 'recent', 1, 12, scope),
      catalogService.getFilters(scope),
    ]);
    res.render('catalog/index', {
      catalog: catalogPage,
      filters,
      userRole: req.session?.user?.role || 'PUBLIC',
    });
  } catch (err) { next(err); }
});

export { router as catalogPageRouter };
```

**public/js/catalog-filter.js — Client-side AJAX filter handler:**

```javascript
(function() {
  'use strict';

  const filterForm = document.getElementById('filter-form');
  const sortSelect = document.getElementById('sort-select');
  const catalogGrid = document.getElementById('catalog-grid');
  const catalogStatus = document.getElementById('catalog-status');
  const clearFiltersBtn = document.getElementById('clear-filters');

  if (!filterForm || !catalogGrid) return;

  let currentPage = 1;

  function buildQueryString(page) {
    const params = new URLSearchParams();
    const formData = new FormData(filterForm);
    for (const [key, value] of formData.entries()) {
      if (value) params.append(key, value);
    }
    if (sortSelect) params.set('sort', sortSelect.value);
    params.set('page', page || 1);
    params.set('page_size', '12');
    return params.toString();
  }

  function renderCards(records) {
    if (records.length === 0) {
      return '<div class="empty-state" role="status"><p>No records match your current filters. ' +
        'Try clearing some filters or <a href="/submit">submit a mission problem</a>.</p></div>';
    }
    return records.map(function(r) {
      const tags = (r.mission_area_tags || []).slice(0, 3)
        .map(function(t) { return '<li class="tag tag--mission">' + escapeHtml(t) + '</li>'; }).join('');
      return [
        '<article class="catalog-card" aria-label="' + escapeHtml(r.title) + '">',
        '  <a href="' + escapeHtml(r.record_url) + '" class="catalog-card__link">',
        '    <header class="catalog-card__header">',
        '      <h3 class="catalog-card__title">' + escapeHtml(r.title) + '</h3>',
        '      <div class="catalog-card__badges">',
        '        <span class="badge badge-maturity">' + escapeHtml(r.maturity_label) + '</span>',
        '        <span class="badge badge-review-status">' + escapeHtml(r.review_status_label) + '</span>',
        '      </div>',
        '    </header>',
        '    <p class="catalog-card__summary">' + escapeHtml(r.short_summary) + '</p>',
        tags ? '<ul class="tag-list">' + tags + '</ul>' : '',
        '  </a>',
        '</article>',
      ].join('\n');
    }).join('\n');
  }

  function escapeHtml(str) {
    if (!str) return '';
    return String(str)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&#39;');
  }

  function updateCatalog(page) {
    currentPage = page || 1;
    const qs = buildQueryString(currentPage);
    fetch('/api/v1/catalog?' + qs)
      .then(function(res) { return res.json(); })
      .then(function(data) {
        // Replace grid content (no full page reload)
        catalogGrid.innerHTML = renderCards(data.records);
        // Announce result count to screen readers via aria-live
        if (catalogStatus) {
          catalogStatus.textContent = 'Showing ' + data.total + ' record' + (data.total === 1 ? '' : 's');
        }
        // Update URL for bookmarkability (does not trigger page reload)
        const newUrl = window.location.pathname + '?' + qs;
        history.pushState({ page: currentPage }, '', newUrl);
      })
      .catch(function(err) {
        console.error('Catalog filter error:', err);
      });
  }

  // Attach listeners — filter changes trigger AJAX update (no form submit / page reload)
  filterForm.addEventListener('change', function() { updateCatalog(1); });
  if (sortSelect) sortSelect.addEventListener('change', function() { updateCatalog(1); });

  if (clearFiltersBtn) {
    clearFiltersBtn.addEventListener('click', function() {
      filterForm.reset();
      updateCatalog(1);
    });
  }

  // Pagination buttons (delegated event on catalog-main)
  document.addEventListener('click', function(e) {
    const btn = e.target.closest('[data-page]');
    if (btn) {
      updateCatalog(parseInt(btn.dataset.page, 10));
    }
  });
})();
```

NOTE: The AJAX-rendered cards are escaped via `escapeHtml()` in catalog-filter.js — this is the client-side XSS guard for dynamically rendered card content.

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-11 | Tampering (stored XSS — server-side) | src/views/catalog/index.html and src/views/partials/catalog-card.html render title, short_summary, tags from DB via EJS | mitigate | All EJS output uses `<%=` (HTML-escaped) for record fields in server-rendered catalog templates — never `<%-` for user-controlled content; verified by grep in verify step |
