**Create src/public/js/perspective-toggle.js:**

```javascript
(function () {
  'use strict';

  function initPerspectiveToggle() {
    var container = document.getElementById('perspective-toggle-container');
    if (!container) return;

    var tabs = Array.from(container.querySelectorAll('[role="tab"]'));
    var panels = {
      EXECUTIVE: document.getElementById('executive-perspective'),
      TECHNICAL: document.getElementById('technical-perspective'),
    };

    function activateTab(perspectiveKey) {
      // Update tabs
      tabs.forEach(function (tab) {
        var isActive = tab.dataset.perspective === perspectiveKey;
        tab.setAttribute('aria-selected', isActive ? 'true' : 'false');
        tab.classList.toggle('perspective-tab--active', isActive);
        tab.setAttribute('tabindex', isActive ? '0' : '-1');
      });
      // Update panels
      Object.keys(panels).forEach(function (key) {
        if (panels[key]) {
          panels[key].style.display = key === perspectiveKey ? 'block' : 'none';
        }
      });
      // Sync URL param without page reload
      var url = new URL(window.location.href);
      url.searchParams.set('view', perspectiveKey.toLowerCase());
      window.history.pushState({ perspective: perspectiveKey }, '', url.toString());
    }

    // Click handler
    tabs.forEach(function (tab) {
      tab.addEventListener('click', function () {
        activateTab(tab.dataset.perspective);
      });
    });

    // Keyboard navigation: arrow keys within tablist (WAI-ARIA tab pattern)
    tabs.forEach(function (tab) {
      tab.addEventListener('keydown', function (e) {
        var currentIndex = tabs.indexOf(e.currentTarget);
        var nextIndex;
        if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
          nextIndex = (currentIndex + 1) % tabs.length;
        } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
          nextIndex = (currentIndex - 1 + tabs.length) % tabs.length;
        } else {
          return;
        }
        e.preventDefault();
        tabs[nextIndex].focus();
        activateTab(tabs[nextIndex].dataset.perspective);
      });
    });

    // Sync on popstate (browser back/forward)
    window.addEventListener('popstate', function (e) {
      if (e.state && e.state.perspective) {
        activateTab(e.state.perspective);
      }
    });
  }

  // Initialize when DOM is ready
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initPerspectiveToggle);
  } else {
    initPerspectiveToggle();
  }
})();
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client→DOM | ?view= URL query parameter from user crosses into EJS template rendering and JS DOM manipulation |
| db→render | No new DB read in this plan — perspective panel content comes from 03-02's record fetch |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-10 | Tampering (XSS via URL param) | ?view= query param reflected into EJS aria-selected / CSS class attribute values | mitigate | In `src/routes/records.ts`, ?view= is normalized to a controlled string: `activeView` is set to either `'EXECUTIVE'` or `'TECHNICAL'` (string literal comparison) or `null` — never the raw query string. EJS renders the boolean expression result `(activeView || defaultPerspective) === 'EXECUTIVE'`, not the raw param value. No user-controlled string is placed in HTML output from this param. |
| T-03-11 | Tampering | perspective-toggle.js calls history.pushState with view param derived from tab's data-perspective attribute | mitigate | `data-perspective` values in DOM are set by EJS template rendering the controlled `EXECUTIVE`/`TECHNICAL` enum from DB (not from query param). `pushState` receives `perspectiveKey.toLowerCase()` where `perspectiveKey` is from `tab.dataset.perspective` — the DOM attribute set by EJS from a DB enum value, not from any untrusted source. No URL construction from user-controlled input. |
| T-03-12 | Information disclosure | Technical perspective section may contain security_findings text from DB — sensitive findings visible to all PUBLIC users | accept | security_findings is treated as public content by design: it is part of the PUBLISHED record surface. Publication workflow (Phase 2 governance gate, curator review) is the control. Curators are responsible for not publishing sensitive findings. Accepted: this is a feature, not a vulnerability. |
