**1. Create src/services/artifact-reachability.ts:**

```typescript
import { AuditService } from './audit';

export interface ReachabilityResult {
  link_id: string;
  url: string;
  reachable: boolean;
  http_status: number | null;
  error: string | null;
}

interface ArtifactLink {
  link_id: string;
  url: string;
  label: string;
  artifact_type: string;
}

const TIMEOUT_MS = 5000; // 5 second timeout per HEAD request
const HTTPS_PATTERN = /^https:\/\//i;

export class ArtifactReachabilityService {
  private auditService: AuditService;

  constructor(auditService?: AuditService) {
    this.auditService = auditService || new AuditService();
  }

  async checkAll(
    links: ArtifactLink[],
    recordId: string,
    userId: string
  ): Promise<ReachabilityResult[]> {
    const results: ReachabilityResult[] = [];

    for (const link of links) {
      // Only check HTTPS URLs; skip non-https (DB constraint should prevent, but guard here)
      if (!HTTPS_PATTERN.test(link.url)) {
        results.push({
          link_id: link.link_id,
          url: link.url,
          reachable: false,
          http_status: null,
          error: 'URL must use HTTPS',
        });
        continue;
      }

      const result = await this.checkOne(link);
      results.push(result);

      // Write warning to audit_log for non-reachable URLs
      if (!result.reachable) {
        try {
          await this.auditService.log(
            'innovation_records',
            recordId,
            'ARTIFACT_URL_WARNING',
            userId,
            null,
            {
              link_id: link.link_id,
              url: link.url,
              artifact_type: link.artifact_type,
              http_status: result.http_status,
              error: result.error,
              checked_at: new Date().toISOString(),
            }
          );
        } catch (auditErr) {
          // Audit log failure must not bubble up — this is fire-and-forget
          console.warn('ArtifactReachabilityService: audit log failed for', link.url, auditErr);
        }
      }
    }

    return results;
  }

  private async checkOne(link: ArtifactLink): Promise<ReachabilityResult> {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), TIMEOUT_MS);

    try {
      const response = await fetch(link.url, {
        method: 'HEAD',
        signal: controller.signal,
        redirect: 'follow',
      });
      clearTimeout(timeoutId);

      return {
        link_id: link.link_id,
        url: link.url,
        reachable: response.status >= 200 && response.status < 400,
        http_status: response.status,
        error: null,
      };
    } catch (err: any) {
      clearTimeout(timeoutId);
      const isTimeout = err.name === 'AbortError';
      return {
        link_id: link.link_id,
        url: link.url,
        reachable: false,
        http_status: null,
        error: isTimeout ? 'Request timed out' : (err.message || 'Network error'),
      };
    }
  }
}
```

if (updatedRecord.artifactLinks && updatedRecord.artifactLinks.length > 0) {

  setImmediate(async () => {

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| curator→external-service | artifact_links.url values from DB are sent as outbound HEAD requests from the server — SSRF risk if URLs are not constrained to HTTPS external resources |
| db→render | artifact_reachability_warnings JSONB serialized into EJS data attribute — XSS if not safely escaped |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-13 | Elevation of privilege (SSRF) | ArtifactReachabilityService.checkOne — server makes outbound HTTP HEAD request to curator-supplied URL | mitigate | `HTTPS_PATTERN` guard in `src/services/artifact-reachability.ts::checkAll` rejects any non-HTTPS URL before calling fetch — prevents requests to http://, file://, ftp://, internal IP ranges via non-HTTPS. Additionally, the DB DDL constraint from Phase 2 (02-01, T-02-05) already rejects non-HTTPS URLs at write time. Both layers enforce HTTPS-only for outbound requests. |
| T-03-14 | Denial of service | ArtifactReachabilityService — HEAD request to slow/hanging server could block event loop | mitigate | `TIMEOUT_MS = 5000` AbortController in `src/services/artifact-reachability.ts::checkOne` — request is aborted after 5 seconds. Additionally, checkAll runs via `setImmediate` (fire-and-forget) so it never blocks the HTTP response path. |
| T-03-15 | Tampering (XSS via JSONB advisory) | artifact_reachability_warnings JSONB rendered into DOM via `data-warnings` attribute in edit.html | mitigate | `JSON.stringify(reachabilityWarnings)` in EJS uses `<%=` (HTML-escaped output in attribute context) in `src/views/admin/records/edit.html`. The advisory text shown in the DOM is built from `warning.error` (a server-controlled string, not a raw URL) using `textContent = ...` (not innerHTML) in the client-side JS, preventing XSS injection. |
| T-03-16 | Information disclosure | ARTIFACT_URL_WARNING audit log entries contain artifact URLs — these are already stored in record_artifact_links; no new sensitive data introduced | accept | The URLs are already in the innovation_records data model. Audit log entries are CURATOR-accessible only (audit API from 02-01 requires requireRole('CURATOR')). Accepted: no new sensitive data surface; audit log is the intended channel for this information. |
</threat_model>
