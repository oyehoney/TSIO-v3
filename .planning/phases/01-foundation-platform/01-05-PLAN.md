**src/services/settings.ts**:
```typescript
import { db } from '../db/knex';

interface HubSettingRow {
  key: string;
  value: string;
  description: string | null;
  updated_by_user_id: string | null;
  updated_at: string;
}

export class SettingsService {
  /**
   * Retrieve a single setting value by key.
   * Returns null if key does not exist.
   */
  async getSetting(key: string): Promise<string | null> {
    const row = await db('hub_settings').where({ key }).first<HubSettingRow>();
    return row?.value ?? null;
  }

  /**
   * Retrieve all hub settings (for admin list view).
   * TechArch: GET /api/v1/admin/settings → list hub_settings (ADMIN)
   */
  async getAllSettings(): Promise<HubSettingRow[]> {
    return db('hub_settings').select('*').orderBy('key', 'asc');
  }

  /**
   * Update a setting value.
   * PLAT-03: Called by ADMIN only (enforced at route layer).
   * Records updated_by_user_id and updated_at.
   *
   * Callers MUST call req.auditService.log() after this to satisfy PLAT-02.
   */
  async updateSetting(key: string, value: string, updatedByUserId: string): Promise<HubSettingRow> {
    const existing = await db('hub_settings').where({ key }).first<HubSettingRow>();
    if (!existing) {
      throw new Error(`Setting key '${key}' does not exist`);
    }

    await db('hub_settings').where({ key }).update({
      value,
      updated_by_user_id: updatedByUserId,
      updated_at: db.fn.now(),
    });

    const updated = await db('hub_settings').where({ key }).first<HubSettingRow>();
    return updated!;
  }
}
```

**src/routes/admin/settings.ts** — TechArch API spec:
```
GET  /api/v1/admin/settings         → list hub_settings (ADMIN)
PUT  /api/v1/admin/settings/:key    → update a setting value (ADMIN)
```

```typescript
import { Router } from 'express';
import { z } from 'zod';
import { requireAuth, requireRole } from '../../middleware/auth';
import { SettingsService } from '../../services/settings';
import type { UserSession } from '../../auth/session';

export const settingsRouter = Router();
const settingsService = new SettingsService();

/**
 * GET /api/v1/admin/settings
 * TechArch: list hub_settings (ADMIN)
 */
settingsRouter.get('/', requireAuth, requireRole('ADMIN'), async (req, res, next) => {
  try {
    const settings = await settingsService.getAllSettings();
    res.json(settings);
  } catch (err) {
    next(err);
  }
});

const updateSettingSchema = z.object({
  value: z.string().min(1, 'Value cannot be empty').max(1000, 'Value too long'),
});

/**
 * PUT /api/v1/admin/settings/:key
 * TechArch: update a setting value (ADMIN)
 * PLAT-02: Writes audit_log entry with before/after values.
 */
settingsRouter.put('/:key', requireAuth, requireRole('ADMIN'), async (req, res, next) => {
  try {
    const { key } = req.params;
    const parseResult = updateSettingSchema.safeParse(req.body);
    if (!parseResult.success) {
      res.status(400).json({ error: parseResult.error.errors[0].message });
      return;
    }

    const { value } = parseResult.data;
    const user = req.session.user as UserSession;

    // Capture before value for audit log
    const beforeSetting = await settingsService.getSetting(key);
    if (beforeSetting === null) {
      res.status(404).json({ error: `Setting '${key}' not found` });
      return;
    }

    const updated = await settingsService.updateSetting(key, value, user.userId);

    // PLAT-02: Audit the settings change
    await req.auditService.log({
      entityType: 'hub_settings',
      entityId: key,   // hub_settings.key is the natural identifier
      action: 'SETTING_UPDATE',
      changedByUserId: user.userId,
      changedByEmail: user.email,
      beforeValue: { key, value: beforeSetting },
      afterValue: { key, value },
      metadata: { setting_key: key },
    });

    res.json({ key: updated.key, value: updated.value });
  } catch (err) {
    next(err);
  }
});

/**
 * GET /api/v1/admin/settings/page
 * Serves the HTML settings management page (server-rendered for WCAG compliance).
 * ADMIN only.
 */
settingsRouter.get('/page', requireAuth, requireRole('ADMIN'), async (req, res, next) => {
  try {
    const settings = await settingsService.getAllSettings();
    // Server-render with current values embedded in the page
    const engagementEmail = settings.find(s => s.key === 'engagement_routing_email')?.value ?? '';
    res.render('admin/settings', {
      settings,
      engagementEmail,
      user: req.session.user,
    });
  } catch (err) {
    next(err);
  }
});
```

**src/routes/admin/audit-log.ts** — TechArch spec:
```
GET /api/v1/admin/audit-log → paginated audit log (CURATOR)
```

```typescript
import { Router } from 'express';
import { z } from 'zod';
import { requireAuth, requireRole } from '../../middleware/auth';
import { AuditService } from '../../services/audit';

export const auditLogRouter = Router();
const auditService = new AuditService();

const querySchema = z.object({
  page: z.coerce.number().int().min(1).default(1),
  pageSize: z.coerce.number().int().min(1).max(100).default(50),
  entityType: z.string().optional(),
  entityId: z.string().uuid().optional(),
});

/**
 * GET /api/v1/admin/audit-log
 * TechArch: paginated audit log (CURATOR)
 * Returns global log or entity-scoped log based on query params.
 */
auditLogRouter.get('/', requireAuth, requireRole('CURATOR'), async (req, res, next) => {
  try {
    const parseResult = querySchema.safeParse(req.query);
    if (!parseResult.success) {
      res.status(400).json({ error: parseResult.error.errors[0].message });
      return;
    }

    const { page, pageSize, entityType, entityId } = parseResult.data;

    let result;
    if (entityType && entityId) {
      result = await auditService.getLog(entityType, entityId, page, pageSize);
    } else {
      result = await auditService.getGlobalLog(page, pageSize);
    }

    res.json({
      rows: result.rows,
      total: result.total,
      page,
      pageSize,
      totalPages: Math.ceil(result.total / pageSize),
    });
  } catch (err) {
    next(err);
  }
});
```
