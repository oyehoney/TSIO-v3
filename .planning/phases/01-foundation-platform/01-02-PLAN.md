        updated_by_user_id      UUID            REFERENCES users(id) ON DELETE SET NULL,
        created_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        updated_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        search_vector           TSVECTOR
    );

    CREATE INDEX idx_records_publication_state ON innovation_records(publication_state);
    CREATE INDEX idx_records_maturity ON innovation_records(maturity_level);
    CREATE INDEX idx_records_mission_area ON innovation_records(mission_area);
    CREATE INDEX idx_records_search ON innovation_records USING GIN(search_vector);
    CREATE INDEX idx_records_last_reviewed ON innovation_records(last_reviewed_at);

    CREATE OR REPLACE FUNCTION update_innovation_record_search_vector()
    RETURNS TRIGGER AS $$
    BEGIN
        NEW.search_vector :=
            setweight(to_tsvector('english', coalesce(NEW.title, '')), 'A') ||
            setweight(to_tsvector('english', coalesce(NEW.short_summary, '')), 'A') ||
            setweight(to_tsvector('english', coalesce(NEW.problem_statement, '')), 'B') ||
            setweight(to_tsvector('english', coalesce(NEW.outcome, '')), 'B') ||
            setweight(to_tsvector('english', coalesce(array_to_string(NEW.technology_areas, ' '), '')), 'C') ||
            setweight(to_tsvector('english', coalesce(NEW.evidence, '')), 'C') ||
            setweight(to_tsvector('english', coalesce(NEW.hypothesis, '')), 'D') ||
            setweight(to_tsvector('english', coalesce(NEW.reuse_guidance, '')), 'D');
        RETURN NEW;
    END;
    $$ LANGUAGE plpgsql;

    CREATE TRIGGER trg_innovation_records_search
        BEFORE INSERT OR UPDATE ON innovation_records
        FOR EACH ROW EXECUTE FUNCTION update_innovation_record_search_vector();

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| migration→db | Migration SQL crosses from application code into the PostgreSQL engine; if parameterization is skipped, injection is possible |
| seed→db | Seed data crosses from application code into PostgreSQL; seed values in Phase 1 are static, but pattern must not evolve to accept user input |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | Migration SQL injection | mitigate | All migration DDL uses `knex.raw()` with static string literals — no dynamic interpolation in migration files. Enforced by code review: grep for `knex.raw(\`.*\${` in db/migrations/* must return 0 matches. |
| T-02-02 | Information disclosure | Sensitive schema detail in error logs | mitigate | Knex connection errors do not echo the DATABASE_URL (which includes password). Winston logger in `src/db/knex.ts` must be set to redact connection strings — implemented via logger sanitizer in Plan 01-03. |
| T-02-03 | Repudiation | Missing audit trail for schema changes | accept | Knex migrations are version-controlled in git; the migration table (`knex_migrations`) records run history. Database-level schema change auditing (DDL audit log) is not required for FISMA Moderate at this scale — git history is the authoritative record. Risk owned by TSIO I&R team. |
| T-02-04 | Elevation of privilege | audit_log append-only constraint violated at DB level | accept | The application enforces append-only semantics (never issues DELETE on audit_log). A true DB-level constraint (e.g., row-level security policy blocking DELETE) would require more infrastructure; deferred to ATO documentation review (Phase 6 COMP-05). Risk owned by TSIO I&R team. |
| T-02-05 | Denial of service | FTS trigger causes write amplification on high-frequency inserts | accept | innovation_records is a low-write table (curators publish records, not end users submitting). At MVP scale (3–5 records at launch, <100 over lifecycle), trigger overhead is negligible. Accepted and documented. |
