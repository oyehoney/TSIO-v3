The audit_log table schema from TechArch (EXACT):
```sql
CREATE TABLE audit_log (
    id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    entity_type         VARCHAR(50)     NOT NULL,
    entity_id           UUID            NOT NULL,
    action              VARCHAR(50)     NOT NULL,
    changed_by_user_id  UUID            REFERENCES users(id) ON DELETE SET NULL,
    changed_by_email    VARCHAR(320),
    before_value        JSONB,
    after_value         JSONB,
    metadata            JSONB,
    created_at          TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);
```

Implementation:
```typescript
import { Knex } from 'knex';
import { db as defaultDb } from '../db/knex';

export interface AuditEntry {
  entityType: string;           // e.g. 'innovation_record', 'hub_settings', 'user'
  entityId: string;             // UUID of the entity being changed
  action: string;               // e.g. 'CREATE', 'UPDATE', 'PUBLISH', 'ARCHIVE', 'SETTING_UPDATE'
  changedByUserId?: string;     // UUID from users.id (null for system/anonymous actions)
  changedByEmail?: string;      // Email preserved even after user deletion (ON DELETE SET NULL)
  beforeValue?: Record<string, unknown>;  // Snapshot of fields BEFORE the change (JSONB)
  afterValue?: Record<string, unknown>;   // Snapshot of fields AFTER the change (JSONB)
  metadata?: Record<string, unknown>;     // Additional context (e.g. IP, reason, transition)
}

export interface AuditRow {
  id: string;
  entity_type: string;
  entity_id: string;
  action: string;
  changed_by_user_id: string | null;
  changed_by_email: string | null;
  before_value: Record<string, unknown> | null;
  after_value: Record<string, unknown> | null;
  metadata: Record<string, unknown> | null;
  created_at: string;
}

export class AuditService {
  private db: Knex;

  constructor(db: Knex = defaultDb) {
    this.db = db;
  }

  /**
   * PLAT-02: Log a material change to any entity.
   * Append-only: this method never deletes or updates audit_log rows.
   * changedByEmail is stored at insert time to preserve identity after user deletion.
   */
  async log(entry: AuditEntry): Promise<void> {
    await this.db('audit_log').insert({
      entity_type: entry.entityType,
      entity_id: entry.entityId,
      action: entry.action,
      changed_by_user_id: entry.changedByUserId ?? null,
      changed_by_email: entry.changedByEmail ?? null,
      before_value: entry.beforeValue ? JSON.stringify(entry.beforeValue) : null,
      after_value: entry.afterValue ? JSON.stringify(entry.afterValue) : null,
      metadata: entry.metadata ? JSON.stringify(entry.metadata) : null,
      // created_at is set by PostgreSQL DEFAULT NOW()
    });
  }

  /**
   * PLAT-02: Retrieve paginated audit log for a specific entity.
   * TechArch API: GET /api/v1/admin/audit-log (CURATOR role)
   * Returns rows ordered by created_at DESC.
   */
  async getLog(
    entityType: string,
    entityId: string,
    page = 1,
    pageSize = 20
  ): Promise<{ rows: AuditRow[]; total: number }> {
    const offset = (page - 1) * pageSize;

    const [{ count }] = await this.db('audit_log')
      .where({ entity_type: entityType, entity_id: entityId })
      .count<[{ count: string }]>('id as count');

    const rows = await this.db('audit_log')
      .where({ entity_type: entityType, entity_id: entityId })
      .orderBy('created_at', 'desc')
      .limit(pageSize)
      .offset(offset)
      .select<AuditRow[]>('*');

    return { rows, total: parseInt(count, 10) };
  }

  /**
   * Retrieve paginated global audit log (all entities) for admin view.
   * Used by GET /api/v1/admin/audit-log
   */
  async getGlobalLog(
    page = 1,
    pageSize = 50
  ): Promise<{ rows: AuditRow[]; total: number }> {
    const offset = (page - 1) * pageSize;

    const [{ count }] = await this.db('audit_log')
      .count<[{ count: string }]>('id as count');

    const rows = await this.db('audit_log')
      .orderBy('created_at', 'desc')
      .limit(pageSize)
      .offset(offset)
      .select<AuditRow[]>('*');

    return { rows, total: parseInt(count, 10) };
  }
}
```

**src/middleware/audit-context.ts** — attaches AuditService to every request for easy access in route handlers:
```typescript
import { Request, Response, NextFunction } from 'express';
import { AuditService } from '../services/audit';
import { db } from '../db/knex';

// Extend Express Request type to include auditService
declare global {
  namespace Express {
    interface Request {
      auditService: AuditService;
    }
  }
}

/**
 * PLAT-02: Attach AuditService to every request.
 * Handlers write audit entries via: req.auditService.log({ ... })
 * Never call AuditService directly in route handlers — always go through this attachment.
 */
export function auditContext(req: Request, _res: Response, next: NextFunction): void {
  req.auditService = new AuditService(db);
  next();
}
```

/**
 * AuditService integration tests.
 * PLAT-02: Proves audit_log inserts, retrieves, and maintains append-only invariant.
 * Runs against real PostgreSQL (docker-compose db service).
 */
import { AuditService } from '../src/services/audit';
import { db } from '../src/db/knex';
import { v4 as uuidv4 } from 'uuid';

let auditService: AuditService;
const TEST_ENTITY_ID = uuidv4();
const TEST_ENTITY_TYPE = 'test_entity';

beforeAll(async () => {
  auditService = new AuditService(db);
});

afterAll(async () => {
  // Clean up test audit rows (test isolation — not a violation of append-only; this is test data)
  await db('audit_log').where({ entity_type: TEST_ENTITY_TYPE }).delete();
  await db.destroy();
});

describe('AuditService.log()', () => {
  test('inserts a row with all required fields', async () => {
    await auditService.log({
      entityType: TEST_ENTITY_TYPE,
      entityId: TEST_ENTITY_ID,
      action: 'CREATE',
      changedByEmail: 'testuser@courts.gov',
      afterValue: { title: 'Test Record', status: 'DRAFT' },
    });

    const row = await db('audit_log')
      .where({ entity_type: TEST_ENTITY_TYPE, entity_id: TEST_ENTITY_ID, action: 'CREATE' })
      .first();

    expect(row).toBeDefined();
    expect(row.entity_type).toBe(TEST_ENTITY_TYPE);
    expect(row.entity_id).toBe(TEST_ENTITY_ID);
    expect(row.action).toBe('CREATE');
    expect(row.changed_by_email).toBe('testuser@courts.gov');
    expect(row.changed_by_user_id).toBeNull();
    expect(row.before_value).toBeNull();
    expect(row.after_value).toMatchObject({ title: 'Test Record', status: 'DRAFT' });
    expect(row.created_at).toBeDefined();
  });

  test('inserts before_value and after_value as JSONB (field-level diff)', async () => {
    await auditService.log({
      entityType: TEST_ENTITY_TYPE,
      entityId: TEST_ENTITY_ID,
      action: 'UPDATE',
      changedByEmail: 'curator@courts.gov',
      beforeValue: { title: 'Old Title', maturity_level: 'IDEA' },
      afterValue: { title: 'New Title', maturity_level: 'EXPERIMENT_POC' },
    });

    const row = await db('audit_log')
      .where({ entity_type: TEST_ENTITY_TYPE, entity_id: TEST_ENTITY_ID, action: 'UPDATE' })
      .first();

    expect(row.before_value).toMatchObject({ title: 'Old Title', maturity_level: 'IDEA' });
    expect(row.after_value).toMatchObject({ title: 'New Title', maturity_level: 'EXPERIMENT_POC' });
  });

  test('stores changedByEmail independently of changedByUserId (non-repudiation)', async () => {
    // Even if user is later deleted (ON DELETE SET NULL on FK), email is preserved
    await auditService.log({
      entityType: TEST_ENTITY_TYPE,
      entityId: TEST_ENTITY_ID,
      action: 'PUBLISH',
      changedByUserId: undefined,   // no FK reference in this test
      changedByEmail: 'preserved-after-deletion@courts.gov',
      afterValue: { publication_state: 'PUBLISHED' },
    });

    const row = await db('audit_log')
      .where({ entity_type: TEST_ENTITY_TYPE, entity_id: TEST_ENTITY_ID, action: 'PUBLISH' })
      .first();

    expect(row.changed_by_email).toBe('preserved-after-deletion@courts.gov');
    expect(row.changed_by_user_id).toBeNull();
  });

  test('stores metadata JSONB (e.g. publication reason, IP address)', async () => {
    await auditService.log({
      entityType: TEST_ENTITY_TYPE,
      entityId: TEST_ENTITY_ID,
      action: 'ARCHIVE',
      changedByEmail: 'admin@courts.gov',
      metadata: { reason: 'Superseded by newer POC', ip: '10.0.0.1' },
      afterValue: { publication_state: 'ARCHIVED' },
    });

    const row = await db('audit_log')
      .where({ entity_type: TEST_ENTITY_TYPE, entity_id: TEST_ENTITY_ID, action: 'ARCHIVE' })
      .first();

    expect(row.metadata).toMatchObject({ reason: 'Superseded by newer POC' });
  });
});

describe('AuditService.getLog()', () => {
  test('returns paginated rows for a specific entity ordered by created_at DESC', async () => {
    const { rows, total } = await auditService.getLog(TEST_ENTITY_TYPE, TEST_ENTITY_ID, 1, 10);

    expect(total).toBeGreaterThanOrEqual(4); // 4 inserts above
    expect(rows.length).toBeGreaterThanOrEqual(4);
    // Ordered DESC — most recent first
    if (rows.length >= 2) {
      expect(new Date(rows[0].created_at).getTime()).toBeGreaterThanOrEqual(
        new Date(rows[rows.length - 1].created_at).getTime()
      );
    }
  });

  test('returns empty rows for non-existent entity', async () => {
    const { rows, total } = await auditService.getLog('nonexistent', uuidv4());
    expect(rows).toHaveLength(0);
    expect(total).toBe(0);
  });

  test('pagination limits results correctly', async () => {
    const { rows } = await auditService.getLog(TEST_ENTITY_TYPE, TEST_ENTITY_ID, 1, 2);
    expect(rows.length).toBeLessThanOrEqual(2);
  });
});

describe('Append-only invariant', () => {
  test('AuditService has no delete() method', () => {
    // Structural check: AuditService must not expose a way to delete audit rows
    const service = new AuditService(db);
    expect(typeof (service as unknown as Record<string, unknown>).delete).toBe('undefined');
    expect(typeof (service as unknown as Record<string, unknown>).truncate).toBe('undefined');
  });
});

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client→audit | Route handlers pass user-supplied field values into before_value/after_value JSONB; attacker may attempt to inject malicious data into the audit log |
| db→render | Audit log rows are read back and rendered in the admin UI (Phase 5); stored JSONB crosses from DB into template rendering |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Tampering | Audit log row deleted or modified by application | mitigate | `AuditService` class has no `delete()` or `update()` method on `audit_log`. Structural test in `tests/audit.test.ts` asserts no delete/truncate methods exist. Additional: `grep -rn "db('audit_log').delete\|db('audit_log').update" src/` in CI must return 0 matches. |
| T-04-02 | Repudiation | changed_by_email not captured, making user actions deniable | mitigate | `AuditService.log()` requires `changedByEmail` to be passed by all callers; the field is stored in `audit_log.changed_by_email` column which survives `ON DELETE SET NULL` on the FK. Future callers in Phase 2+ MUST pass `req.session.user?.email` — enforced by TypeScript interface (field is optional but code-reviewed). |
| T-04-03 | Information disclosure | JSONB before_value/after_value contains PII logged in plaintext | accept | The audit log is access-controlled (CURATOR role required for GET /api/v1/admin/audit-log). Fields stored in audit_log are the same fields curators enter — no additional PII beyond what's in the record. PostgreSQL storage encryption (COMP-04) protects at rest. Risk owned by TSIO I&R team per ATO documentation. |
| T-04-04 | Denial of service | Audit log table grows unbounded — no purge policy | accept | At MVP scale (3–5 records, handful of curators) audit_log growth is negligible. A retention/archival policy is deferred to Phase 6 ATO documentation (COMP-05). Risk owned by TSIO I&R team. |
| T-04-05 | Tampering | Malicious JSONB content injected via before_value/after_value | mitigate | `AuditService.log()` calls `JSON.stringify()` on before/after values before passing to Knex, ensuring the DB receives well-formed JSONB, not raw interpolated SQL. Knex parameterizes the INSERT — no SQL injection path. Verified by: `grep -n 'JSON.stringify' src/services/audit.ts`. |
