// FRD DDL (Phase 1 — already created, reference only):
// CREATE TABLE opportunity_submissions (
//     submission_id       UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
//     problem_description TEXT        NOT NULL CHECK (LENGTH(problem_description) >= 50),
//     mission_area        VARCHAR(200) NOT NULL,
//     submitting_office   VARCHAR(200) NOT NULL,
//     submitter_name      VARCHAR(200) NOT NULL,
//     submitter_email     VARCHAR(255) NOT NULL,
//     submitter_title     VARCHAR(200),
//     urgency_context     TEXT,
//     known_constraints   TEXT,
//     status              VARCHAR(30) NOT NULL DEFAULT 'SUBMITTED' CHECK (status IN (
//                             'SUBMITTED', 'UNDER_REVIEW', 'ACCEPTED_FOR_CONSIDERATION',
//                             'DECLINED', 'LINKED_TO_RECORD'
//                         )),
//     disposition         VARCHAR(30),
//     linked_record_id    UUID        REFERENCES innovation_records(record_id),
//     internal_note       TEXT,
//     submitted_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
//     reviewed_at         TIMESTAMPTZ,
//     reviewed_by_user_id UUID        REFERENCES users(user_id)
// );

import { db } from '../db/knex';

export interface OpportunitySubmission {
  submission_id: string;
  problem_description: string;
  mission_area: string;
  submitting_office: string;
  submitter_name: string;
  submitter_email: string;
  submitter_title?: string;
  urgency_context?: string;
  known_constraints?: string;
  status: string;
  disposition?: string;
  linked_record_id?: string;
  internal_note?: string;
  submitted_at: Date;
  reviewed_at?: Date;
  reviewed_by_user_id?: string;
}

export class SubmissionRepository {
  async insert(data: Omit<OpportunitySubmission, 'submission_id' | 'submitted_at'>): Promise<OpportunitySubmission> {
    const [row] = await db('opportunity_submissions').insert(data).returning('*');
    return row;
  }

  async findById(submission_id: string): Promise<OpportunitySubmission | undefined> {
    return db('opportunity_submissions').where({ submission_id }).first();
  }

  async list(filters: { status?: string; limit?: number; offset?: number }): Promise<OpportunitySubmission[]> {
    let query = db('opportunity_submissions').orderBy('submitted_at', 'desc');
    if (filters.status) query = query.where({ status: filters.status });
    if (filters.limit) query = query.limit(filters.limit);
    if (filters.offset) query = query.offset(filters.offset);
    return query;
  }

  async update(submission_id: string, patch: Partial<Pick<OpportunitySubmission, 'status' | 'disposition' | 'linked_record_id' | 'internal_note' | 'reviewed_at' | 'reviewed_by_user_id'>>): Promise<OpportunitySubmission> {
    const [row] = await db('opportunity_submissions')
      .where({ submission_id })
      .update(patch)
      .returning('*');
    return row;
  }
}

### Step 3: src/services/SubmissionService.ts
```typescript
import { SubmissionRepository } from '../repositories/SubmissionRepository';
import { verifyCaptcha } from '../lib/captcha';
import { sendSubmissionNotification } from '../lib/mailer';

// Simple HTML tag stripper (no external lib required for MVP)
function stripHtml(input: string): string {
  return input.replace(/<[^>]*>/g, '').trim();
}

// Simple email validator
function isValidEmail(email: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

const VALID_STATUSES = ['SUBMITTED', 'UNDER_REVIEW', 'ACCEPTED_FOR_CONSIDERATION', 'DECLINED', 'LINKED_TO_RECORD'] as const;

export class SubmissionService {
  constructor(private repo = new SubmissionRepository()) {}

  async createSubmission(body: {
    problem_description: string;
    mission_area: string;
    submitting_office: string;
    submitter_name: string;
    submitter_email: string;
    submitter_title?: string;
    urgency_context?: string;
    known_constraints?: string;
    captcha_token: string;
  }) {
    // 1. Verify captcha
    const captchaOk = await verifyCaptcha(body.captcha_token);
    if (!captchaOk) throw Object.assign(new Error('CAPTCHA_INVALID'), { code: 'CAPTCHA_INVALID', status: 422 });

    // 2. Strip HTML from problem_description (server-side)
    const problem_description = stripHtml(body.problem_description);

    // 3. Validate all fields per FRD validation rules
    if (!problem_description || problem_description.length < 50 || problem_description.length > 3000) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422,
        detail: 'problem_description must be 50–3,000 characters' });
    }
    if (!body.mission_area || body.mission_area.length < 2 || body.mission_area.length > 200) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    if (!body.submitting_office || body.submitting_office.length < 2 || body.submitting_office.length > 200) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    if (!body.submitter_name || body.submitter_name.length < 2 || body.submitter_name.length > 200) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    if (!body.submitter_email || !isValidEmail(body.submitter_email)) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    if (body.submitter_title && body.submitter_title.length > 200) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    if (body.urgency_context && body.urgency_context.length > 1000) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    if (body.known_constraints && body.known_constraints.length > 1000) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }

    // 4. Insert submission
    const submission = await this.repo.insert({
      problem_description,
      mission_area: body.mission_area,
      submitting_office: body.submitting_office,
      submitter_name: body.submitter_name,
      submitter_email: body.submitter_email,
      submitter_title: body.submitter_title,
      urgency_context: body.urgency_context,
      known_constraints: body.known_constraints,
      status: 'SUBMITTED',
    });

    // 5. Email — soft-fail
    await sendSubmissionNotification(submission);

    return submission;
  }

  async listSubmissions(filters: { status?: string; limit?: number; offset?: number }) {
    return this.repo.list(filters);
  }

  async updateSubmission(submission_id: string, patch: {
    status?: string; internal_note?: string; linked_record_id?: string; disposition?: string;
  }, actorId: string) {
    // Validate: LINKED_TO_RECORD requires linked_record_id
    if (patch.status === 'LINKED_TO_RECORD' && !patch.linked_record_id) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422,
        detail: 'linked_record_id is required when status is LINKED_TO_RECORD' });
    }
    return this.repo.update(submission_id, {
      ...patch,
      reviewed_at: new Date(),
      reviewed_by_user_id: actorId,
    });
  }
}

// FRD F05 API Endpoints:
// POST   /api/v1/opportunity-submissions        — PUBLIC, no auth
// GET    /api/v1/opportunity-submissions        — CURATOR
// PATCH  /api/v1/opportunity-submissions/:id   — CURATOR

import { Router } from 'express';
import { SubmissionService } from '../../services/SubmissionService';
import { requireAuth, requireRole } from '../../middleware/auth';
import { submissionRateLimit } from '../../middleware/rate-limit';

const router = Router();
const svc = new SubmissionService();

// POST — PUBLIC, rate limited
router.post('/', submissionRateLimit, async (req, res) => {
  try {
    const result = await svc.createSubmission(req.body);
    return res.status(201).json(result);
  } catch (err: any) {
    return res.status(err.status ?? 500).json({ error: err.code ?? 'INTERNAL_ERROR', detail: err.detail });
  }
});

// GET — CURATOR only
router.get('/', requireAuth, requireRole('CURATOR'), async (req, res) => {
  const { status, limit = '50', offset = '0' } = req.query as Record<string, string>;
  const results = await svc.listSubmissions({ status, limit: parseInt(limit, 10), offset: parseInt(offset, 10) });
  return res.json(results);
});

// PATCH — CURATOR only
router.patch('/:id', requireAuth, requireRole('CURATOR'), async (req, res) => {
  try {
    const actorId = (req.user as any).user_id;
    const result = await svc.updateSubmission(req.params.id, req.body, actorId);
    return res.json(result);
  } catch (err: any) {
    return res.status(err.status ?? 500).json({ error: err.code ?? 'INTERNAL_ERROR', detail: err.detail });
  }
});

export { router as opportunitySubmissionsRouter };
```

import { Router } from 'express';
import { SubmissionService } from '../../services/SubmissionService';
import { submissionRateLimit } from '../../middleware/rate-limit';

const router = Router();
const svc = new SubmissionService();

// GET /submit-opportunity — render public form (no auth required)
router.get('/', (req, res) => {
  res.render('public/submit-opportunity', {
    title: 'Submit an Opportunity | TSIO Innovation Hub',
    recaptchaSiteKey: process.env.RECAPTCHA_SITE_KEY ?? '',
    errors: null,
    formData: {},
  });
});

// POST /submit-opportunity — form handler (HTML form submission, rate limited)
// captcha_token injected by reCAPTCHA widget into form body via hidden field
router.post('/', submissionRateLimit, async (req, res) => {
  try {
    await svc.createSubmission(req.body);
    return res.redirect('/submit-opportunity/confirmation');
  } catch (err: any) {
    // Re-render form with error state (not a separate API call — this is server-rendered form)
    return res.status(422).render('public/submit-opportunity', {
      title: 'Submit an Opportunity | TSIO Innovation Hub',
      recaptchaSiteKey: process.env.RECAPTCHA_SITE_KEY ?? '',
      errors: [err.detail ?? err.message ?? 'Submission failed. Please try again.'],
      formData: req.body,
    });
  }
});

// GET /submit-opportunity/confirmation — success page
router.get('/confirmation', (req, res) => {
  res.render('public/submit-opportunity-confirmation', {
    title: 'Submission Received | TSIO Innovation Hub',
  });
});

export { router as submitOpportunityRouter };
