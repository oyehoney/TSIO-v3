// src/lib/captcha.ts
import axios from 'axios';

export async function verifyCaptcha(token: string): Promise<boolean> {
  // Dev bypass: allow 'BYPASS' token when NODE_ENV is 'development' or 'test'
  if ((process.env.NODE_ENV === 'development' || process.env.NODE_ENV === 'test')
      && token === 'BYPASS') {
    return true;
  }
  const secret = process.env.RECAPTCHA_SECRET_KEY;
  if (!secret) {
    console.warn('[captcha] RECAPTCHA_SECRET_KEY not set; rejecting token');
    return false;
  }
  try {
    const response = await axios.post(
      'https://www.google.com/recaptcha/api/siteverify',
      null,
      { params: { secret, response: token } }
    );
    return response.data.success === true && response.data.score >= 0.5;
  } catch (err) {
    console.warn('[captcha] verification request failed:', err);
    return false;
  }
}

export async function sendEngagementNotification(request: {
  request_id: string; record_id: string; request_type: string;
  requestor_name: string; requestor_email: string; requestor_office: string;
  description_of_interest: string;
}): Promise<void> {
  try {
    const to = await getRoutingEmail();

    const transporter = createTransporter();
    await transporter.sendMail({
      to,
      subject: `[TSIO Hub] New Engagement Request: ${request.request_type}`,

        `From:       ${request.requestor_name} <${request.requestor_email}> (${request.requestor_office})`,

    });

import { db } from '../db/knex';

export interface EngagementRequest {
  request_id: string;
  record_id: string;
  request_type: string;
  requestor_name: string;
  requestor_email: string;
  requestor_office: string;
  requestor_title?: string;
  description_of_interest: string;
  desired_next_step?: string;
  status: string;
  curator_note?: string;
  submitted_at: Date;
  updated_at: Date;
  updated_by_user_id?: string;
}

export class EngagementRepository {
  async insert(data: Omit<EngagementRequest, 'request_id' | 'submitted_at' | 'updated_at'>): Promise<EngagementRequest> {
    const [row] = await db('engagement_requests').insert(data).returning('*');
    return row;
  }

  async findById(request_id: string): Promise<EngagementRequest | undefined> {
    return db('engagement_requests').where({ request_id }).first();
  }

  async list(filters: { record_id?: string; status?: string; limit?: number; offset?: number }): Promise<EngagementRequest[]> {
    let query = db('engagement_requests').orderBy('submitted_at', 'desc');
    if (filters.record_id) query = query.where({ record_id: filters.record_id });
    if (filters.status) query = query.where({ status: filters.status });
    if (filters.limit) query = query.limit(filters.limit);
    if (filters.offset) query = query.offset(filters.offset);
    return query;
  }

  async update(request_id: string, patch: { status?: string; curator_note?: string; updated_by_user_id?: string }): Promise<EngagementRequest> {
    const [row] = await db('engagement_requests')
      .where({ request_id })
      .update({ ...patch, updated_at: db.fn.now() })
      .returning('*');
    return row;
  }
}

### Step 5: src/services/EngagementService.ts
Business logic — validates record published status, validates request_type against
`record_engagement_options` table (Phase 1 DDL), verifies captcha, sends email:

```typescript
import { db } from '../db/knex';
import { EngagementRepository } from '../repositories/EngagementRepository';
import { verifyCaptcha } from '../lib/captcha';
import { sendEngagementNotification } from '../lib/mailer';

const VALID_TYPES = ['REQUEST_DEMO', 'REQUEST_ADOPTION_DISCUSSION', 'REQUEST_TECHNICAL_GUIDANCE', 'REQUEST_BRIEFING'] as const;

export class EngagementService {
  constructor(private repo = new EngagementRepository()) {}

  async createEngagementRequest(body: {
    record_id: string; request_type: string; requestor_name: string;
    requestor_email: string; requestor_office: string; requestor_title?: string;
    description_of_interest: string; desired_next_step?: string; captcha_token: string;
  }) {
    // 1. Validate captcha
    const captchaOk = await verifyCaptcha(body.captcha_token);
    if (!captchaOk) throw Object.assign(new Error('CAPTCHA_INVALID'), { code: 'CAPTCHA_INVALID', status: 422 });

    // 2. Validate record exists and is PUBLISHED
    const record = await db('innovation_records')
      .where({ record_id: body.record_id, publication_state: 'PUBLISHED' })
      .first();
    if (!record) throw Object.assign(new Error('RECORD_NOT_FOUND'), { code: 'RECORD_NOT_FOUND', status: 404 });

    // 3. Validate request_type against record_engagement_options
    const option = await db('record_engagement_options')
      .where({ record_id: body.record_id, option_type: body.request_type, is_enabled: true })
      .first();
    if (!option) throw Object.assign(new Error('INVALID_ENGAGEMENT_TYPE'), { code: 'INVALID_ENGAGEMENT_TYPE', status: 422 });

    // 4. Validate description_of_interest length (20-2000 chars)
    if (!body.description_of_interest || body.description_of_interest.length < 20) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }

    // 5. Insert
    const request = await this.repo.insert({
      record_id: body.record_id,
      request_type: body.request_type,
      requestor_name: body.requestor_name,
      requestor_email: body.requestor_email,
      requestor_office: body.requestor_office,
      requestor_title: body.requestor_title,
      description_of_interest: body.description_of_interest,
      desired_next_step: body.desired_next_step,
      status: 'SUBMITTED',
      updated_by_user_id: undefined,
    });

    // 6. Email — soft-fail (sendEngagementNotification catches internally)
    await sendEngagementNotification(request);

    return request;
  }

  async listEngagementRequests(filters: { record_id?: string; status?: string; limit?: number; offset?: number }) {
    return this.repo.list(filters);
  }

  async updateEngagementRequest(request_id: string, patch: { status?: string; curator_note?: string }, actorId: string) {
    if (patch.curator_note && patch.curator_note.length > 500) {
      throw Object.assign(new Error('VALIDATION_ERROR'), { code: 'VALIDATION_ERROR', status: 422 });
    }
    return this.repo.update(request_id, { ...patch, updated_by_user_id: actorId });
  }
}
```

### Step 6: src/routes/api/engagement-requests.ts

```typescript
// Implements (from FRD — F07):
// POST   /api/v1/engagement-requests        — PUBLIC, no auth
// GET    /api/v1/engagement-requests        — CURATOR (supports ?record_id= filter)
// PATCH  /api/v1/engagement-requests/:id    — CURATOR

import { Router } from 'express';
import { EngagementService } from '../../services/EngagementService';
import { requireAuth, requireRole } from '../../middleware/auth';
import { engagementRateLimit } from '../../middleware/rate-limit';

const router = Router();
const svc = new EngagementService();

// POST /api/v1/engagement-requests — PUBLIC, no auth, rate limited
router.post('/', engagementRateLimit, async (req, res) => {
  try {
    const result = await svc.createEngagementRequest(req.body);
    return res.status(201).json(result);
  } catch (err: any) {
    const status = err.status ?? 500;
    return res.status(status).json({ error: err.code ?? 'INTERNAL_ERROR' });
  }
});

// GET /api/v1/engagement-requests — CURATOR only
router.get('/', requireAuth, requireRole('CURATOR'), async (req, res) => {
  const { record_id, status, limit = '50', offset = '0' } = req.query as Record<string, string>;
  const results = await svc.listEngagementRequests({
    record_id, status,
    limit: parseInt(limit, 10),
    offset: parseInt(offset, 10),
  });
  return res.json(results);
});

// PATCH /api/v1/engagement-requests/:id — CURATOR only
router.patch('/:id', requireAuth, requireRole('CURATOR'), async (req, res) => {
  try {
    const actorId = (req.user as any).user_id;
    const result = await svc.updateEngagementRequest(req.params.id, req.body, actorId);
    return res.json(result);
  } catch (err: any) {
    const status = err.status ?? 500;
    return res.status(status).json({ error: err.code ?? 'INTERNAL_ERROR' });
  }
});

export { router as engagementRequestsRouter };
```
