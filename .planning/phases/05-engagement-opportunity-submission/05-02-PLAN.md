**Step 5: Create src/public/js/engagement-modal.js**

Vanilla JavaScript (no framework — consistent with EJS stack). Module pattern:

```javascript
(function () {
  'use strict';

  var modal = document.getElementById('engagement-modal');
  var form = document.getElementById('engagement-modal-form');
  var closeBtn = document.getElementById('engagement-modal-close');
  var doneBtn = document.getElementById('engagement-modal-done');
  var confirmation = document.getElementById('engagement-modal-confirmation');
  var errorsDiv = document.getElementById('engagement-modal-errors');
  var submitBtn = document.getElementById('engagement-submit-btn');
  var SITE_KEY = document.querySelector('meta[name="recaptcha-site-key"]')
    ? document.querySelector('meta[name="recaptcha-site-key"]').getAttribute('content')
    : '';

  // Open modal on engagement button click
  document.querySelectorAll('.engagement-trigger').forEach(function (btn) {
    btn.addEventListener('click', function () {
      openModal(
        btn.getAttribute('data-record-id'),
        btn.getAttribute('data-request-type'),
        btn.getAttribute('data-record-title')
      );
    });
  });

  function openModal(recordId, requestType, recordTitle) {
    document.getElementById('modal-record-id').value = recordId;
    document.getElementById('modal-request-type').value = requestType;
    document.getElementById('engagement-modal-title').textContent =
      'Request: ' + requestType.replace(/_/g, ' ').replace(/^REQUEST /, '');
    form.hidden = false;
    confirmation.hidden = true;
    errorsDiv.hidden = true;
    errorsDiv.innerHTML = '';
    modal.hidden = false;
    // Focus first focusable element inside modal
    var firstFocusable = modal.querySelector('button, input, textarea, select, [tabindex]');
    if (firstFocusable) firstFocusable.focus();
    // Trap focus
    modal.addEventListener('keydown', trapFocus);
  }

  function closeModal() {
    modal.hidden = true;
    modal.removeEventListener('keydown', trapFocus);
    form.reset();
    confirmation.hidden = true;
    errorsDiv.hidden = true;
    // Return focus to trigger button
    var trigger = document.querySelector('.engagement-trigger[data-request-type="' +
      document.getElementById('modal-request-type').value + '"]');
    if (trigger) trigger.focus();
  }

  // Escape key and close button
  closeBtn.addEventListener('click', closeModal);
  if (doneBtn) doneBtn.addEventListener('click', closeModal);
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && !modal.hidden) closeModal();
  });

  function trapFocus(e) {
    if (e.key !== 'Tab') return;
    var focusable = modal.querySelectorAll(
      'button:not([hidden]), input:not([hidden]), textarea, select, [tabindex]:not([tabindex="-1"])'
    );
    var first = focusable[0];
    var last = focusable[focusable.length - 1];
    if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
    else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
  }

  // Form submission
  form.addEventListener('submit', async function (e) {
    e.preventDefault();
    submitBtn.disabled = true;
    submitBtn.textContent = 'Submitting...';
    errorsDiv.hidden = true;
    errorsDiv.innerHTML = '';

    try {
      // Inject reCAPTCHA v3 token before sending
      var token = '';
      if (typeof grecaptcha !== 'undefined' && SITE_KEY) {
        token = await grecaptcha.execute(SITE_KEY, { action: 'engage' });
      } else {
        token = 'BYPASS'; // dev/test bypass
      }
      document.getElementById('modal-captcha-token').value = token;

      var formData = new FormData(form);
      var body = {};
      formData.forEach(function (val, key) { body[key] = val; });

      var response = await fetch('/api/v1/engagement-requests', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
      });

      if (response.ok) {
        // Show exact FRD confirmation text
        form.hidden = true;
        confirmation.hidden = false;
        var confirmMsg = confirmation.querySelector('p');
        if (confirmMsg) {
          confirmMsg.textContent =
            'Your request has been sent to the I\u0026R team. Someone will follow up with you based on team availability.';
        }
      } else {
        var data = await response.json();
        showErrors([data.error || 'Submission failed. Please try again.']);
        if (response.status === 429) {
          showErrors(['You have submitted too many requests. Please try again later.']);
        }
      }
    } catch (err) {
      showErrors(['An error occurred. Please try again.']);
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Submit Request';
    }
  });

  function showErrors(messages) {
    errorsDiv.innerHTML = '<ul>' + messages.map(function (m) {
      return '<li>' + escapeHtml(m) + '</li>';
    }).join('') + '</ul>';
    errorsDiv.hidden = false;
    errorsDiv.focus();
  }

  function escapeHtml(str) {
    return str
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }
})();
```
