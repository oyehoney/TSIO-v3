**tests/integration/performance-baseline.test.js**

```javascript
'use strict';
/**
 * Performance Baseline Test
 *
 * Measures p95 latency for GET /api/v1/catalog against a running PostgreSQL
 * instance with 3 seeded records. Passes if p95 < 3000ms.
 *
 * Note: These are sequential measurements, not truly concurrent (Jest runs
 * in a single process). This measures per-request latency under no load.
 * For concurrent load testing, see docs/PERFORMANCE.md — k6/autocannon
 * instructions for manual verification.
 *
 * Skipped if DATABASE_URL is not set.
 */

const request = require('supertest');
const knex = require('knex');
const { createApp } = require('../../src/app');

const SKIP_DB = !process.env.DATABASE_URL;
const REQUEST_COUNT = 20; // Run 20 requests, compute p95 (19th percentile of sorted)
const P95_THRESHOLD_MS = 3000;

let db, app;

beforeAll(async () => {
  if (SKIP_DB) return;
  db = knex({ client: 'pg', connection: process.env.DATABASE_URL, pool: { min: 1, max: 5 } });
  app = createApp({ db });
});

afterAll(async () => { if (db) await db.destroy(); });

(SKIP_DB ? describe.skip : describe)('Performance Baseline — GET /api/v1/catalog', () => {
  test(`p95 latency < ${P95_THRESHOLD_MS}ms over ${REQUEST_COUNT} sequential requests`, async () => {
    const latencies = [];

    // Warm-up (2 requests, not measured)
    await request(app).get('/api/v1/catalog');
    await request(app).get('/api/v1/catalog');

    // Measured requests
    for (let i = 0; i < REQUEST_COUNT; i++) {
      const start = Date.now();
      const res = await request(app).get('/api/v1/catalog');
      const elapsed = Date.now() - start;
      expect(res.status).toBe(200);
      latencies.push(elapsed);
    }

    // Compute percentiles
    const sorted = [...latencies].sort((a, b) => a - b);
    const p50 = sorted[Math.floor(REQUEST_COUNT * 0.50)];
    const p95 = sorted[Math.floor(REQUEST_COUNT * 0.95)];
    const p99 = sorted[Math.floor(REQUEST_COUNT * 0.99)] || sorted[sorted.length - 1];
    const min = sorted[0];
    const max = sorted[sorted.length - 1];
    const avg = Math.round(latencies.reduce((a, b) => a + b, 0) / latencies.length);

    console.log(`\n=== PERFORMANCE BASELINE RESULTS ===`);
    console.log(`Endpoint: GET /api/v1/catalog`);
    console.log(`Requests: ${REQUEST_COUNT} sequential (after 2 warm-up)`);
    console.log(`Min: ${min}ms | Avg: ${avg}ms | p50: ${p50}ms | p95: ${p95}ms | p99: ${p99}ms | Max: ${max}ms`);
    console.log(`Threshold: p95 < ${P95_THRESHOLD_MS}ms`);
    console.log(`Result: ${p95 < P95_THRESHOLD_MS ? 'PASS ✓' : 'FAIL ✗'}`);
    console.log(`=====================================\n`);

    expect(p95).toBeLessThan(P95_THRESHOLD_MS);
  }, 60000); // 60 second timeout for 20 requests

  test('GET /api/v1/records/:id p95 < 3000ms over 10 sequential requests', async () => {
    const AUDIO_SEC_ID = '11111111-1111-1111-1111-111111111001';
    const latencies = [];

    // Warm-up
    await request(app).get(`/api/v1/records/${AUDIO_SEC_ID}`);

    for (let i = 0; i < 10; i++) {
      const start = Date.now();
      const res = await request(app).get(`/api/v1/records/${AUDIO_SEC_ID}`);
      const elapsed = Date.now() - start;
      expect(res.status).toBe(200);
      latencies.push(elapsed);
    }

    const sorted = [...latencies].sort((a, b) => a - b);
    const p95 = sorted[Math.floor(10 * 0.95)] || sorted[sorted.length - 1];
    console.log(`\nRecord page p95: ${p95}ms (threshold: ${P95_THRESHOLD_MS}ms)`);
    expect(p95).toBeLessThan(P95_THRESHOLD_MS);
  }, 30000);
});
```
