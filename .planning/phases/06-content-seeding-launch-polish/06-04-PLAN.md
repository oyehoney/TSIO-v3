**tests/integration/wcag-accessibility.test.js**

Verifies that all API responses include the fields required for WCAG-compliant frontend rendering. These tests encode the data contract the frontend depends on for accessible rendering.

```javascript
'use strict';
/**
 * WCAG Accessibility Data Contract Tests
 *
 * This project is a JSON REST API. WCAG 2.1 AA compliance for a backend
 * means ensuring all fields required for accessible UI rendering are present
 * in every API response. A frontend that renders without these fields cannot
 * be accessible.
 *
 * Required fields for accessible rendering:
 * - maturity_label (human-readable, for screen readers — not just the enum value)
 * - review_status_label (same)
 * - trust_disclaimers (required on every published record — COMP-03 + F9)
 * - mission_area_tags and technology_area_tags as string arrays (for filter/context)
 *
 * Catalog card fields (for accessible catalog card rendering):
 * - maturity_level (for badge color via BADGE_COLORS map)
 * - maturity_label (for screen reader label alongside badge)
 * - record_id (for navigation links)
 * - title (for card heading)
 * - outcome_summary (for card body — must be non-null for PUBLISHED records)
 */

const request = require('supertest');
const knex = require('knex');
const { createApp } = require('../../src/app');

const AUDIO_SEC_ID = '11111111-1111-1111-1111-111111111001';
const AI_REDACT_ID = '11111111-1111-1111-1111-111111111002';

const SKIP_DB = !process.env.DATABASE_URL;

let db, publicApp;

beforeAll(async () => {
  if (SKIP_DB) return;
  db = knex({ client: 'pg', connection: process.env.DATABASE_URL, pool: { min: 1, max: 5 } });
  publicApp = createApp({ db });
});

afterAll(async () => {
  if (db) await db.destroy();
});

describe('Innovation Record API — WCAG required fields', () => {
  (SKIP_DB ? test.skip : test)(
    'GET /api/v1/records/:id includes all WCAG-required fields for accessible rendering',
    async () => {
      const res = await request(publicApp).get(`/api/v1/records/${AUDIO_SEC_ID}`);
      expect(res.status).toBe(200);

      // Fields required for accessible UI rendering
      expect(res.body.maturity_label).toBeTruthy();           // e.g. "Experiment / POC"
      expect(res.body.review_status_label).toBeTruthy();      // e.g. "Technically Reviewed"
      expect(res.body.title).toBeTruthy();                    // heading
      expect(res.body.problem_statement).toBeTruthy();        // body content
      expect(res.body.outcome_summary).toBeTruthy();          // body content

      // Trust disclaimers — required on every PUBLISHED record (COMP-03 + F9)
      expect(Array.isArray(res.body.trust_disclaimers)).toBe(true);
      expect(res.body.trust_disclaimers.length).toBeGreaterThan(0);
      // Each disclaimer must be a non-empty string (for screen reader rendering)
      res.body.trust_disclaimers.forEach(d => {
        expect(typeof d).toBe('string');
        expect(d.length).toBeGreaterThan(0);
      });

      // Tag arrays (for accessible filter and context rendering)
      expect(Array.isArray(res.body.mission_area_tags)).toBe(true);
      expect(Array.isArray(res.body.technology_area_tags)).toBe(true);

      // Maturity level enum (for badge color lookup)
      expect(['IDEA', 'EXPERIMENT_POC', 'PROTOTYPE_PILOT', 'PRODUCTION_VALIDATED', 'ARCHIVED'])
        .toContain(res.body.maturity_level);

      // Review status enum
      expect(['SUBMITTED', 'CURATED', 'TECHNICALLY_REVIEWED', 'SECURITY_REVIEWED',
               'POLICY_REVIEWED', 'VALIDATED_FOR_REUSE', 'SUPERSEDED_RETIRED'])
        .toContain(res.body.review_status);

      // Perspective fields (required for perspective toggle accessibility)
      expect(res.body.executive_perspective_text).toBeTruthy();
      expect(['EXECUTIVE', 'TECHNICAL']).toContain(res.body.default_perspective);

      // Key findings (required for findings list rendering)
      expect(Array.isArray(res.body.key_findings)).toBe(true);

      // Artifact links (required for accessible external link rendering)
      expect(Array.isArray(res.body.artifact_links)).toBe(true);

      // Engagement options (required for accessible NextActionPanel rendering)
      expect(Array.isArray(res.body.engagement_options)).toBe(true);
    }
  );

  (SKIP_DB ? test.skip : test)(
    'GET /api/v1/records/:id — EXPERIMENT_POC record has at least 2 trust_disclaimers (Rules 1 + 2)',
    async () => {
      const res = await request(publicApp).get(`/api/v1/records/${AUDIO_SEC_ID}`);
      expect(res.status).toBe(200);
      expect(res.body.maturity_level).toBe('EXPERIMENT_POC');
      // Rule 1 fires (EXPERIMENT_POC) + Rule 2 fires (PUBLISHED) = minimum 2
      expect(res.body.trust_disclaimers.length).toBeGreaterThanOrEqual(2);
    }
  );
});

describe('Catalog API — WCAG required fields for card rendering', () => {
  (SKIP_DB ? test.skip : test)(
    'GET /api/v1/catalog includes maturity_level, maturity_label, and record_id for each card',
    async () => {
      const res = await request(publicApp).get('/api/v1/catalog');
      expect(res.status).toBe(200);

      // Handle both array and paginated response shapes
      const items = res.body.items || res.body.data || (Array.isArray(res.body) ? res.body : []);
      expect(items.length).toBeGreaterThan(0);

      items.forEach(card => {
        expect(card.record_id).toBeTruthy();            // for navigation links
        expect(card.title).toBeTruthy();                // for card heading
        expect(card.maturity_level).toBeTruthy();       // for badge color lookup
        // maturity_label may or may not be in catalog cards — check at least maturity_level
        expect(['IDEA', 'EXPERIMENT_POC', 'PROTOTYPE_PILOT', 'PRODUCTION_VALIDATED', 'ARCHIVED'])
          .toContain(card.maturity_level);
      });
    }
  );

  (SKIP_DB ? test.skip : test)(
    'GET /api/v1/catalog — all cards have non-null outcome_summary for accessible preview text',
    async () => {
      const res = await request(publicApp).get('/api/v1/catalog');
      expect(res.status).toBe(200);
      const items = res.body.items || res.body.data || (Array.isArray(res.body) ? res.body : []);
      items.forEach(card => {
        // outcome_summary is required for catalog card body text (accessible preview)
        expect(card.outcome_summary || card.short_summary).toBeTruthy();
      });
    }
  );
});

describe('Badge colors — WCAG 2.1 AA contrast', () => {
  test('All maturity badge colors pass WCAG 2.1 AA 4.5:1 contrast ratio', () => {
    const { verifyBadgeContrast, BADGE_COLORS } = require('../../src/services/trustDisclaimerService');

    expect(BADGE_COLORS).toBeDefined();
    expect(BADGE_COLORS.EXPERIMENT_POC).toBe('#D97706');
    expect(BADGE_COLORS.PROTOTYPE_PILOT).toBe('#EA580C');
    expect(BADGE_COLORS.PRODUCTION_VALIDATED).toBe('#16A34A');
    expect(BADGE_COLORS.ARCHIVED).toBe('#374151');
    expect(BADGE_COLORS.IDEA).toBe('#6B7280');

    const results = verifyBadgeContrast();
    results.forEach(r => {
      expect(r.passesAA).toBe(true, `${r.maturityLevel} (${r.color}) fails WCAG AA: white=${r.contrastAgainstWhite.toFixed(2)}, black=${r.contrastAgainstBlack.toFixed(2)}`);
    });
  });
});
```

/**
