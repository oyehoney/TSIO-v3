```javascript
'use strict';
const request = require('supertest');
const knex = require('knex');
const { createApp } = require('../../src/app');

const AUDIO_SEC_ID = '11111111-1111-1111-1111-111111111001';
const AI_REDACT_ID = '11111111-1111-1111-1111-111111111002';
const BLOCKCHAIN_ID = '11111111-1111-1111-1111-111111111003';

let db;
let publicApp;

beforeAll(async () => {
  db = knex({ client: 'pg', connection: process.env.DATABASE_URL, pool: { min: 1, max: 5 } });
  publicApp = createApp({ db });
  // Note: seed data must already exist (seeded by 01 and 02 seed scripts before this test runs)
});

afterAll(async () => {
  await db.destroy();
});

describe('Seeded records — API scoping', () => {
  test('AI Redaction POC (PUBLISHED) returns 200 for unauthenticated PUBLIC caller', async () => {
    const res = await request(publicApp).get(`/api/v1/records/${AI_REDACT_ID}`);
    expect(res.status).toBe(200);
    expect(res.body.record_id).toBe(AI_REDACT_ID);
    expect(res.body.trust_disclaimers).toBeInstanceOf(Array);
    expect(res.body.trust_disclaimers.length).toBeGreaterThanOrEqual(2); // Rules 1 + 2
  });

  test('Blockchain Experiment (ARCHIVED) returns 404 for unauthenticated PUBLIC caller', async () => {
    const res = await request(publicApp).get(`/api/v1/records/${BLOCKCHAIN_ID}`);
    expect(res.status).toBe(404);
  });

  test('GET /api/v1/catalog includes both PUBLISHED seeded records', async () => {
    const res = await request(publicApp).get('/api/v1/catalog');
    expect(res.status).toBe(200);
    const ids = (res.body.items || res.body.data || res.body).map(r => r.record_id);
    expect(ids).toContain(AUDIO_SEC_ID);
    expect(ids).toContain(AI_REDACT_ID);
  });

  test('GET /api/v1/catalog does NOT include ARCHIVED seeded record', async () => {
    const res = await request(publicApp).get('/api/v1/catalog');
    expect(res.status).toBe(200);
    const ids = (res.body.items || res.body.data || res.body).map(r => r.record_id);
    expect(ids).not.toContain(BLOCKCHAIN_ID);
  });

  test('DB has exactly 2 PUBLISHED and 1 ARCHIVED seeded records', async () => {
    const rows = await db('innovation_records')
      .whereIn('record_id', [AUDIO_SEC_ID, AI_REDACT_ID, BLOCKCHAIN_ID])
      .select('publication_state');
    const published = rows.filter(r => r.publication_state === 'PUBLISHED').length;
    const archived = rows.filter(r => r.publication_state === 'ARCHIVED').length;
    expect(published).toBe(2);
    expect(archived).toBe(1);
  });
});
```

Note: The catalog response shape may vary (the CatalogService returns `{ data: [...], total, page, pageSize }` — handle both `res.body.items` and `res.body.data` and `res.body` array patterns defensively.
