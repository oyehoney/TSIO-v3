const knex = require('knex');

const SEED_RECORD_ID = '11111111-1111-1111-1111-111111111001';
const SEED_USER_EMAIL = 'seed-curator@ao.uscourts.gov';

async function seed(db) {
  // 1. Upsert seed user (idempotent)
  const [{ user_id: seedUserId }] = await db.raw(`
    INSERT INTO users (email, display_name, role)
    VALUES (?, 'Seed Curator', 'CURATOR')
    ON CONFLICT (email) DO UPDATE SET display_name = EXCLUDED.display_name
    RETURNING user_id
  `, [SEED_USER_EMAIL]).then(r => r.rows);

  // 2. Insert parent record (ON CONFLICT (record_id) DO NOTHING — idempotent)
  const result = await db.raw(`
    INSERT INTO innovation_records (
      record_id, title, problem_statement, what_was_explored, outcome_summary,
      reuse_guidance, short_summary, maturity_level, review_status, reuse_potential,
      source_type, owner_name, owner_office, contributing_office, contributor_attribution,
      executive_perspective_text, executive_recommendation,
      technical_perspective_text, security_findings, performance_findings,
      default_perspective, publication_state, last_reviewed_date, published_at,
      created_by_user_id, updated_by_user_id
    ) VALUES (
      ?, ?, ?, ?, ?,
      ?, ?, ?, ?, ?,
      ?, ?, ?, ?, ?,
      ?, ?,
      ?, ?, ?,
      ?, ?, ?, NOW(),
      ?, ?
    )
    ON CONFLICT (record_id) DO NOTHING
  `, [
    SEED_RECORD_ID,
    // ... all values in order
  ]);

  // 3. Insert child rows only if parent was newly inserted
  if (result.rowCount === 1) {
    // Insert key_findings, artifact_links, record_tags, engagement_options
  }
}

async function main() {
  const db = knex({ client: 'pg', connection: process.env.DATABASE_URL });
  try {
    await seed(db);
    console.log('Audio Security POC seed complete.');
  } finally {
    await db.destroy();
  }
}

main().catch(err => { console.error(err); process.exit(1); });

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| env→process | DATABASE_URL environment variable crossing into Knex connection config — an attacker who can set env vars could redirect the seed to a different database |
| seed→db | Hardcoded content strings crossing into parameterized Knex INSERT — must never use string interpolation |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01 | Tampering | `db/seeds/01_audio_security_poc.js` — hardcoded seed content modified post-code-review | accept | Seed content is committed source code under git history; changes require code review. Risk accepted — this is a dev/deploy-time script, not a runtime attack surface. Owner: development team via git. |
| T-06-02 | Information disclosure | Seed script logs DATABASE_URL in error output | mitigate | Catch block in `main()` only logs `err.message` (not the full connection object); `db.destroy()` in finally block prevents connection leak. Implemented in `main().catch(err => { console.error(err.message); process.exit(1); })`. |
| T-06-03 | Tampering | SQL injection via seed content values crossing into INSERT | mitigate | All values passed as parameterized array to `db.raw('INSERT ... VALUES (?, ?, ...)', [...values])` — Knex never string-interpolates the values array into the query string. Verified by grep: `grep -n 'db.raw' db/seeds/01_audio_security_poc.js` must show parameterized form only. |
| T-06-04 | Elevation of privilege | Seed user (seed-curator@ao.uscourts.gov) persists with CURATOR role in production | accept | Seed script is a dev/CI tool; production deployments should run only approved seed scripts. Accepted: the seed user has CURATOR role with no elevated admin privileges. Documented in OPEN-RISKS.md (Plan 06-05). |
